- The code is chosen with repetition.
- directoryPath must be passed into the function (ex: /res/asset/)

import java.io.File;
import java.util.Random;

public class RandomFilePicker { //if put inside a function then just copy the function pickRandomFXML()
    public static File pickRandomFXML(String directoryPath) {
        File directory = new File(directoryPath);

        if (!directory.exists() || !directory.isDirectory()) {
            System.out.println("Directory not found!");
            return null;
        }

        // Get all .fxml files in the directory
        File[] fxmlFiles = directory.listFiles((dir, name) -> name.endsWith(".fxml"));

        if (fxmlFiles == null || fxmlFiles.length == 0) {
            System.out.println("No .fxml files found in the directory!");
            return null;
        }

        // Generate a random index
        Random random = new Random();
        int randomIndex = random.nextInt(fxmlFiles.length);

        // Return the randomly selected file
        return fxmlFiles[randomIndex];
    }


    //here is an example to call the function
    public static void main(String[] args) {
        String directoryPath = "src/application/"; // Adjust path based on your project structure
        File randomFile = pickRandomFXML(directoryPath);

        if (randomFile != null) {
            System.out.println("Randomly selected file: " + randomFile.getName());
        }
    }
}
